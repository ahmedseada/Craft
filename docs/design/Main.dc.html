using System.Text; using Idrak.Abstraction.Generation; namespace Craft.Plugins.Tools; /// <summary>The editor's tools, offered to any Idrak chat model.</summary> public sealed class EditorTools(IEditorHost host) : IToolProvider { public IReadOnlyList<ITool> Tools => ToolProviders.FromMethods(this); [Tool("get_selection", "The selected text of the active document.")] public string GetSelection() { var document = host.ActiveDocument; return document.Snapshot.Slice(document.Selection).ToString(); } [Tool("get_diagnostics", "Errors and warnings of the active document.")] public async Task<string> GetDiagnosticsAsync(CancellationToken cancellationToken) { var items = await host.DiagnosticsAsync(host.ActiveDocument, cancellationToken); return string.Join('\n', items.Select(d => $"{d.Line}: {d.Message}")); } }
